public abstract class User {
        // ■■ Private Attributes (Encapsulation) ■■
        private int userId;
        private String name;
        private String email;
        private String passwordHash;
        private String mobile;
        private String role;
        private String createdAt;
 
        // ■■ Constructor ■■
        public User(int userId, String name, String email, String passwordHash, String mobile,
String role) {
                this.userId = userId;
                this.name = name;
                this.email = email;
                this.passwordHash = passwordHash;
                this.mobile = mobile;
                this.role = role;
        }
 
        // ■■ Getters & Setters ■■
        public int getUserId() { return userId; }
        public void setUserId(int userId) { this.userId = userId; }
        public String getName() { return name; }
        public void setName(String name) { this.name = name; }
        public String getEmail() { return email; }
        public void setEmail(String email) { this.email = email; }
        public String getMobile() { return mobile; }
        public void setMobile(String mobile) { this.mobile = mobile; }
        public String getRole() { return role; }
        public String getCreatedAt() { return createdAt; }
        public void setCreatedAt(String date) { this.createdAt = date; }
 
        // ■■ Abstract Methods (Polymorphism) ■■
        public abstract boolean login(String email, String password);
        public abstract void displayProfile();
 
        // ■■ Utility ■■
        protected String hashPassword(String password) {
                // Simple hash simulation
                return Integer.toHexString(password.hashCode());
        }
}

                this.walletBalance += amount;
                System.out.println("Wallet topped up. New balance: " + walletBalance);
        }

                bookingHistory.add(bookingId);

                System.out.println("Booking history for " + getName() + ":");
                for (String b : bookingHistory) System.out.println(" - " + b);
        }

        public void saveToFile() throws IOException {
                FileWriter fw = new FileWriter("passengers.txt", true);
                fw.write(getUserId() + "," + getName() + "," + getEmail() + ","
                              + getMobile() + "," + passengerType + "\n");
                fw.close();
        }
 
        // ■■ Load all passengers from file ■■
        public static List loadFromFile() throws IOException {
                List list = new ArrayList<>();
                BufferedReader br = new BufferedReader(new FileReader("passengers.txt"));
                String line;
                while ((line = br.readLine()) != null) {
                        String[] parts = line.split(",");
                        if (parts.length >= 5) {
                                Passenger p = new Passenger(Integer.parseInt(parts[0]),
                                        parts[1], parts[2], "", parts[3]);
                                p.setPassengerType(parts[4]);
                                list.add(p);
                        }
                }
                br.close();
                return list;
        }
 
        @Override
        public boolean login(String email, String password) {
                return this.getEmail().equals(email)
                        && hashPassword(password).equals(this.getMobile()); // simplified check
        }

                System.out.println("Mobile : " + getMobile());

public class PremiumPassenger extends Passenger {
        // ■■ Private Attributes ■■

        private String membershipExpiry;
        private String membershipTier; // 'Gold', 'Platinum'
 
        // ■■ Constructor ■■
        public PremiumPassenger(int userId, String name, String email, String password, String

                super(userId, name, email, password, mobile);
                setPassengerType("Premium");
                this.discountRate = 0.10;
                this.loyaltyPoints = 0;
                this.membershipTier = "Gold";
        }
 
        // ■■ Getters & Setters ■■
        public double getDiscountRate() { return discountRate; }
        public void setDiscountRate(double rate) { this.discountRate = rate; }
        public int getLoyaltyPoints() { return loyaltyPoints; }
        public String getMembershipTier() { return membershipTier; }
        public void setMembershipTier(String tier) { this.membershipTier = tier; }
        public String getMembershipExpiry() { return membershipExpiry; }
        public void setMembershipExpiry(String d) { this.membershipExpiry = d; }
 
        // ■■ Methods ■■
        public double applyDiscount(double fare) {
                double discounted = fare - (fare * discountRate);
                System.out.println("Discount applied: LKR " + (fare - discounted));
                return discounted;
        }
 
        public void addLoyaltyPoints(int points) {
                this.loyaltyPoints += points;
                System.out.println("Points added! Total: " + loyaltyPoints);
                upgradeTier();
        }
 
        private void upgradeTier() {
                if (loyaltyPoints >= 1000) membershipTier = "Platinum";
                else if (loyaltyPoints >= 500) membershipTier = "Gold";
        }
 
        public void redeemPoints(int points) {
                if (points <= loyaltyPoints) {
                        loyaltyPoints -= points;
                        double credit = points * 0.5; // 1 point = LKR 0.50
                        topUpWallet(credit);
                        System.out.println("Redeemed " + points + " points for LKR " + credit);
                } else {
                        System.out.println("Not enough loyalty points.");
                }
        }
 
        @Override
        public void displayProfile() {
                super.displayProfile();
                System.out.println("Tier : " + membershipTier);
                System.out.println("Points : " + loyaltyPoints);
                System.out.println("Discount: " + (discountRate * 100) + "%");
        }
}

class FareCalculator extends (Abstract)
public abstract class FareCalculator {
        protected double baseFare;
        protected double ratePerKm;
        protected double surgeMultiplier;
 
        public double getBaseFare() { return baseFare; }
        public double getRatePerKm() { return ratePerKm; }
        public double getSurgeMultiplier() { return surgeMultiplier; }
        public void setSurgeMultiplier(double surge){ this.surgeMultiplier = surge; }
 
        // Abstract — each ride type overrides (Polymorphism)
        public abstract double calculateFare(double distanceKm);
 
        // Abstraction — hides complex logic from UI
        protected double calculateDistance(String pickup, String dropoff) {
                // Simplified distance calculation
                return Math.abs(pickup.hashCode() - dropoff.hashCode()) % 30 + 1;
        }
 
        public void printFareBreakdown(double distanceKm) {
                System.out.println("Base Fare : LKR " + baseFare);
                System.out.println("Distance : " + distanceKm + " km");
                System.out.println("Rate/km : LKR " + ratePerKm);
                System.out.println("Surge : x" + surgeMultiplier);
                System.out.println("Total Fare : LKR " + calculateFare(distanceKm));
        }
}

public class CashPayment extends Payment {

class RatingSummary
public class RatingSummary {
        private int userId;
        private double averageScore;
        private int totalReviews;
        private int fiveStars, fourStars, threeStars, twoStars, oneStar;
 
        public RatingSummary(int userId) { this.userId = userId; }
 
        public int getUserId() { return userId; }
        public double getAverageScore() { return averageScore; }
        public int getTotalReviews() { return totalReviews; }
 
        public void addRating(int stars) {
                totalReviews++;
                switch(stars) {
                        case 5: fiveStars++; break;
                        case 4: fourStars++; break;
                        case 3: threeStars++; break;
                        case 2: twoStars++; break;
                        case 1: oneStar++; break;
                }
                recalculateAverage();
        }
 
        private void recalculateAverage() {
                int total = (5*fiveStars)+(4*fourStars)+(3*threeStars)+(2*twoStars)+(1*oneStar);
                averageScore = totalReviews > 0 ? (double) total / totalReviews : 0.0;
        }
 
        public double getPercentage(int stars) {
                return totalReviews > 0 ? (double) stars / totalReviews * 100 : 0;
        }
 
        public void displaySummary() {
                System.out.println("Rating Summary for User " + userId);
                System.out.println("Average : " + String.format("%.1f",averageScore) + "/5");
                System.out.println("Total : " + totalReviews + " reviews");
                System.out.println("5 stars : " + fiveStars);
                System.out.println("4 stars : " + fourStars);
                System.out.println("3 stars : " + threeStars);
                System.out.println("2 stars : " + twoStars);
                System.out.println("1 star : " + oneStar);
        }
}
